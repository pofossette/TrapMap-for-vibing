# Operations Route Refactoring — Wave 1: Core Extraction

**Objective:** Extract all 15 route handlers from `operations.ts` into 9 sub-modules under `operations/` directory, creating a thin router that registers sub-routes.

## Module Structure (from CONTEXT.md decisions)

```
packages/server/src/routes/
├── operations.ts              # Thin router (~40 lines) — imports and registers sub-routes
└── operations/
    ├── index.ts               # Barrel export for all sub-routes
    ├── audit.ts               # GET /v1/operations/audit (~35 lines)
    ├── knowledge-legacy.ts    # GET /v1/operations/knowledge + POST /:entryId/deactivate (~200 lines)
    ├── artifacts-import.ts    # POST /v1/operations/import + POST /artifacts/import (~300 lines)
    ├── artifacts-export.ts    # POST /v1/operations/export + POST /artifacts/export (~210 lines)
    ├── artifacts-activate.ts  # POST /v1/operations/artifacts/activate + POST /:artifactId/deactivate (~240 lines)
    ├── migrate.ts             # POST /v1/operations/migrate (~240 lines)
    ├── status.ts              # GET /v1/operations/status (~90 lines)
    ├── skill-edit.ts          # POST /artifacts/:artifactId/edit + GET /:artifactId/history (~220 lines)
    └── skill-review.ts        # GET /artifacts/review-queue + POST /:artifactId/review (~240 lines)
```

import type { FastifyPluginAsync } from 'fastify';

import {
  auditRoutes,
  knowledgeLegacyRoutes,
  artifactsImportRoutes,
  artifactsExportRoutes,
  artifactsActivateRoutes,
  migrateRoutes,
  statusRoutes,
  skillEditRoutes,
  skillReviewRoutes,
} from './operations/index.js';

export const operationsRoutes: FastifyPluginAsync = async (app) => {
  // Register all operation sub-routes
  // Order matters: more specific paths should be registered before parameterized ones
  await app.register(auditRoutes);
  await app.register(knowledgeLegacyRoutes);
  await app.register(artifactsExportRoutes);
  await app.register(artifactsImportRoutes);
  await app.register(artifactsActivateRoutes);
  await app.register(migrateRoutes);
  await app.register(statusRoutes);
  await app.register(skillEditRoutes);
  await app.register(skillReviewRoutes);
};
