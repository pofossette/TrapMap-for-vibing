```typescript
import { describe, expect, it } from 'vitest';

import {
  formatBoundaryExtractionPrompt,
  formatClaimVerificationPrompt,
  formatKnowledgeRefinementPrompt,
  getAllPromptMetadata,
  getPromptMetadata,
  PROMPT_METADATA,
} from './templates.js';
import type { PromptTemplateId } from './types.js';

describe('prompt templates', () => {
  // ===========================================================================
  // Metadata Tests
  // ===========================================================================

  describe('metadata', () => {
    it('has metadata for all three template IDs', () => {
      const ids: PromptTemplateId[] = [
        'claim-verification',
        'boundary-extraction',
        'knowledge-refinement',
      ];

      for (const id of ids) {
        const meta = getPromptMetadata(id);
        expect(meta).toBeDefined();
        expect(meta.id).toBe(id);
        expect(meta.version).toMatch(/^\d+\.\d+\.\d+$/);
        expect(meta.description).toBeTruthy();
        expect(meta.lastModified).toMatch(/^\d{4}-\d{2}-\d{2}T/);
      }
    });

    it('getAllPromptMetadata returns all three templates', () => {
      const all = getAllPromptMetadata();
      expect(all).toHaveLength(3);
      expect(all.map((m) => m.id)).toContain('claim-verification');
      expect(all.map((m) => m.id)).toContain('boundary-extraction');
      expect(all.map((m) => m.id)).toContain('knowledge-refinement');
    });

    it('PROMPT_METADATA is frozen/immutable', () => {
      // Verify the metadata object exists and has expected keys
      expect(Object.keys(PROMPT_METADATA)).toHaveLength(3);
    });
  });

  // ===========================================================================
  // Claim Verification Tests
  // ===========================================================================

  describe('formatClaimVerificationPrompt', () => {
    it('returns [systemPrompt, userMessage] tuple', () => {
      const [system, user] = formatClaimVerificationPrompt({
        context: ['Context 1'],
        claims: [{ text: 'Claim 1' }],
      });

      expect(typeof system).toBe('string');
      expect(typeof user).toBe('string');
      expect(system.length).toBeGreaterThan(0);
      expect(user.length).toBeGreaterThan(0);
    });

    it('system prompt describes claim verification role', () => {
      const [system] = formatClaimVerificationPrompt({
        context: ['Context'],
        claims: [{ text: 'Claim' }],
      });

      expect(system).toContain('claim verification');
    });

    it('user message includes formatted context', () => {
      const [_, user] = formatClaimVerificationPrompt({
        context: ['First context', 'Second context'],
        claims: [{ text: 'Claim' }],
      });

      expect(user).toContain('[1] First context');
      expect(user).toContain('[2] Second context');
    });

    it('user message includes formatted claims', () => {
      const [_, user] = formatClaimVerificationPrompt({
        context: ['Context'],
        claims: [{ text: 'First claim' }, { text: 'Second claim' }],
      });

      expect(user).toContain('1. First claim');
      expect(user).toContain('2. Second claim');
    });

    it('handles empty context array', () => {
      const [system, user] = formatClaimVerificationPrompt({
        context: [],
        claims: [{ text: 'Claim' }],
      });

      expect(system).toBeTruthy();
      expect(user).toContain('Context:');
    });

    it('handles empty claims array', () => {
      const [system, user] = formatClaimVerificationPrompt({
        context: ['Context'],
        claims: [],
      });

      expect(system).toBeTruthy();
      expect(user).toContain('Claims to verify:');
    });

    it('includes JSON response format instruction', () => {
      const [_, user] = formatClaimVerificationPrompt({
        context: ['Context'],
        claims: [{ text: 'Claim' }],
      });

      expect(user).toContain('"verifications"');
      expect(user).toContain('"supported"');
      expect(user).toContain('"evidence"');
    });
  });

  // ===========================================================================
  // Boundary Extraction Tests
  // ===========================================================================

  describe('formatBoundaryExtractionPrompt', () => {
    it('returns [systemPrompt, userMessage] tuple', () => {
      const [system, user] = formatBoundaryExtractionPrompt({
        shortcut: 'React Hooks',
        detail: 'Use useEffect carefully',
        labels: ['react', 'hooks'],
      });

      expect(typeof system).toBe('string');
      expect(typeof user).toBe('string');
      expect(system.length).toBeGreaterThan(0);
      expect(user.length).toBeGreaterThan(0);
    });

    it('system prompt describes boundary extraction role', () => {
      const [system] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'Detail',
        labels: [],
      });

      expect(system).toContain('boundary extraction');
    });

    it('system prompt describes all six boundary layers', () => {
      const [system] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'Detail',
        labels: [],
      });

      expect(system).toContain('context');
      expect(system).toContain('versions');
      expect(system).toContain('prerequisites');
      expect(system).toContain('signals');
      expect(system).toContain('exclusions');
      expect(system).toContain('evidence');
    });

    it('user message includes shortcut (title)', () => {
      const [_, user] = formatBoundaryExtractionPrompt({
        shortcut: 'React Hooks Pitfall',
        detail: 'Detail',
        labels: [],
      });

      expect(user).toContain('Title: React Hooks Pitfall');
    });

    it('user message includes detail', () => {
      const [_, user] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'When using useEffect, remember to handle cleanup',
        labels: [],
      });

      expect(user).toContain('When using useEffect, remember to handle cleanup');
    });

    it('user message includes labels as comma-separated list', () => {
      const [_, user] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'Detail',
        labels: ['react', 'hooks', 'frontend'],
      });

      expect(user).toContain('Labels: react, hooks, frontend');
    });

    it('handles empty labels array', () => {
      const [_, user] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'Detail',
        labels: [],
      });

      expect(user).toContain('Labels:');
    });

    it('includes JSON structure in system prompt', () => {
      const [system] = formatBoundaryExtractionPrompt({
        shortcut: 'Test',
        detail: 'Detail',
        labels: [],
      });

      expect(system).toContain('"context"');
      expect(system).toContain('"versions"');
      expect(system).toContain('"prerequisites"');
      expect(system).toContain('"signals"');
      expect(system).toContain('"exclusions"');
      expect(system).toContain('"evidence"');
    });
  });

  // ===========================================================================
  // Knowledge Refinement Tests
  // ===========================================================================

  describe('formatKnowledgeRefinementPrompt', () => {
    it('returns [systemPrompt, userMessage] tuple', () => {
      const [system, user] = formatKnowledgeRefinementPrompt({
        query: 'How to use React hooks?',
        searchResults: 'Result 1: Use useState for state',
      });

      expect(typeof system).toBe('string');
      expect(typeof user).toBe('string');
      expect(system.length).toBeGreaterThan(0);
      expect(user.length).toBeGreaterThan(0);
    });

    it('system prompt describes knowledge refinement role', () => {
      const [system] = formatKnowledgeRefinementPrompt({
        query: 'Query',
        searchResults: 'Results',
      });

      expect(system).toContain('knowledge refinement');
    });

    it('system prompt includes 3-sentence constraint', () => {
      const [system] = formatKnowledgeRefinementPrompt({
        query: 'Query',
        searchResults: 'Results',
      });

      expect(system).toContain('3 sentences');
    });

    it('user message is the search results', () => {
      const [_, user] = formatKnowledgeRefinementPrompt({
        query: 'Query',
        searchResults: 'Knowledge: React hooks are powerful',
      });

      expect(user).toBe('Knowledge: React hooks are powerful');
    });

    it('handles empty search results', () => {
      const [system, user] = formatKnowledgeRefinementPrompt({
        query: 'Query',
        searchResults: '',
      });

      expect(system).toBeTruthy();
      expect(user).toBe('');
    });

    it('handles multiline search results', () => {
      const results = `Result 1: First item
Result 2: Second item
Result 3: Third item`;

      const [_, user] = formatKnowledgeRefinementPrompt({
        query: 'Query',
        searchResults: results,
      });

      expect(user).toBe(results);
    });
  });
});
```
