    Follow the established pattern from `backfill-indexes.ts` exactly:

    ```typescript
    import type { Pool } from 'pg';
    import { PgCandidateRepository } from '../candidates/pg-repository.js';
    import type { SkillShareerStore } from '../store.js';

    export interface MigrationConfig {
      /** PostgreSQL connection pool */
      pool: Pool;
      /** Store to read candidates from */
      store: SkillShareerStore;
      /** If true, don't write to database, just report what would be done */
      dryRun?: boolean;
      /** Progress callback */
      onProgress?: (info: {
        processed: number;
        total: number;
        candidateId: string;
      }) => void;
    }

    export interface MigrationError {
      candidateId: string;
      error: string;
    }

    export interface MigrationResult {
      /** Total candidates examined */
      totalCandidates: number;
      /** Candidates successfully migrated */
      migrated: number;
      /** Candidates skipped (already exist in table) */
      skipped: number;
      /** Errors encountered */
      errors: MigrationError[];
      /** Total duration in milliseconds */
      durationMs: number;
    }

    export async function migrateCandidates(config: MigrationConfig): Promise<MigrationResult> {
      const { pool, store, dryRun = false, onProgress } = config;

      const startTime = Date.now();
      const result: MigrationResult = {
        totalCandidates: 0,
        migrated: 0,
        skipped: 0,
        errors: [],
        durationMs: 0,
      };

      const data = await store.snapshot();
      const candidates = data.candidateSubmissions;
      result.totalCandidates = candidates.length;

      if (candidates.length === 0) {
        result.durationMs = Date.now() - startTime;
        return result;
      }

      const repo = new PgCandidateRepository(pool);

      // Ensure table exists before migration
      // (PgCandidateRepository.ensureSchema is called internally on first operation)

      for (let i = 0; i < candidates.length; i++) {
        const candidate = candidates[i];
        if (!candidate) continue;

        try {
          if (dryRun) {
            result.skipped++;
          } else {
            // Check if already migrated (idempotent)
            const existing = await repo.getById(candidate.id);
            if (existing) {
              result.skipped++;
            } else {
              await repo.insert(candidate);
              result.migrated++;
            }
          }

          onProgress?.({
            processed: i + 1,
            total: result.totalCandidates,
            candidateId: candidate.id,
          });
        } catch (error) {
          result.errors.push({
            candidateId: candidate.id,
            error: error instanceof Error ? error.message : String(error),
          });
        }
      }

      result.durationMs = Date.now() - startTime;
      return result;
    }
    ```

    Key design decisions:
    - Uses PgCandidateRepository directly (not the DualWriteCandidateRepository) because during migration the data is being read FROM JSONB and written TO the relational table -- the dual-write would try to write back to JSONB which is redundant
    - Idempotent: checks getById() first to skip already-migrated candidates
    - Continues on error: records the error and proceeds with remaining candidates
    - Reports progress via callback for large datasets
