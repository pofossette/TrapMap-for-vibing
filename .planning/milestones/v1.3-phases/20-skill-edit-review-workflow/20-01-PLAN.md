app.post('/v1/knowledge/review', async (request) => {
  const auth = await resolveAuthContext(app.skillShareer, request);
  requirePermission(auth, 'knowledge:review');
  const payload = reviewDecisionRequestSchema.parse(request.body);

  const reviewedEntry = await app.skillShareer.store.transact((data) => {
    const entry = data.knowledgeEntries.find((candidate) => candidate.id === payload.entryId);
    if (entry.teamId) requireTeamAccess(auth, entry.teamId);
    requireHigherLevel(auth, entry.requiredLevel);

    applyReviewDecision({ store, data, entry, reviewerUserId, decidedAt, decision, notes });

    const auditEvent = createAuditEvent({ store, data, teamId, actor, action: 'knowledge-reviewed', entityId, payload });

    return toKnowledgeEntry(data, entry);
  });

    await runKnowledgeIndexEvent({ ... });

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| CLI caller -> server route | Untrusted review payloads and artifact IDs enter the authenticated review endpoint |
| Route -> artifact model | Authorized requests cross from HTTP/auth parsing into artifact state mutation |
| Server response -> CLI formatter | Only filtered metadata should reach human or JSON output |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-20-01 | S | `GET /v1/operations/artifacts/review-queue` | mitigate | Resolve auth from the saved session and require `knowledge:review` before returning any data. |
| T-20-02 | S | `POST /v1/operations/artifacts/:artifactId/review` | mitigate | Require real user (not system-admin) for review decisions to ensure accountability. |
| T-20-03 | I | review queue filtering | mitigate | Only show artifacts where reviewer has strictly higher security level than the artifact's requiredLevel. |
| T-20-04 | T | lifecycle state transition | mitigate | Review decisions are recorded in reviewHistory, lifecycleHistory, and audit trail for traceability. |
| T-20-05 | D | team-scoped artifacts | mitigate | Apply requireTeamAccess() for team-scoped artifacts to prevent cross-team review access. |
| T-20-06 | R | audit trail | accept | All review operations create audit events for traceability. |
</threat_model>
