export interface RagLogEntry {
  timestamp: string;

4. **loadRagLogConfig function:**
```typescript
export function loadRagLogConfig(): RagLogConfig {
  const enabled = process.env.LOG_RAG_ENABLED === 'true';
  const logDir = process.env.LOG_RAG_DIR ?? 'logs/rag';
  return { enabled, logDir };
}
```

5. **logRagRetrieval function:**
```typescript
export async function logRagRetrieval(
  config: RagLogConfig,
  entry: RagLogEntry,
): Promise<void>
```
- Early return if `config.enabled === false`
- Use `mkdir` with `{ recursive: true }` to ensure log directory exists
- Build daily log file path: `path.join(config.logDir, \`${dateStr}.log\`)` where dateStr is YYYY-MM-DD from entry.timestamp
- Format entry as JSON line: `JSON.stringify(entry) + '\n'`
- Append to file with `appendFile(logFile, line, 'utf-8')`
- Wrap in try/catch, log errors to console with `[rag-log]` prefix, do NOT rethrow (fire-and-forget)

6. **generateQueryId helper:**
```typescript
export function generateQueryId(): string {
  return `qry_${Date.now()}_${Math.random().toString(36).slice(2, 8)}`;
}
```
