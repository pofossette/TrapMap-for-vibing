```typescript
/**
 * Apply a manual resolution to a candidate.
 * Triggers the final publish or merge action based on the stored manual result.
 */
export async function applyResolution(options: {
  candidateId: string;
  apiBaseUrl?: string;
  token: string;
  json?: boolean;
}): Promise<void> {
  const baseUrl = options.apiBaseUrl ?? process.env.TRAPMAP_API_URL ?? 'http://localhost:3000';
  const url = `${baseUrl}/v1/candidates/${options.candidateId}/apply-resolution`;

  const response = await fetch(url, {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${options.token}`,
      'Content-Type': 'application/json',
    },
  });

  if (!response.ok) {
    const error = await response.json().catch(() => ({ message: response.statusText }));
    throw new Error(`Failed to apply resolution: ${error.message || response.status}`);
  }

  const result = await response.json();

  if (options.json) {
    console.log(JSON.stringify(result, null, 2));
    return;
  }

  console.log(`✅ Resolution applied successfully`);
  console.log(`   Candidate: ${result.candidateId}`);
  console.log(`   Status: ${result.status}`);
  console.log(`   Decision: ${result.outcome.decision}`);

  if (result.outcome.decision === 'independent') {
    console.log(`   Published as: ${result.outcome.entityType} (${result.outcome.publishedEntityId})`);
  } else {
    console.log(`   Merged into: ${result.outcome.entityType} (${result.outcome.mergedIntoEntityId})`);
  }

  if (result.lineage) {
    console.log(`   Lineage ID: ${result.lineage.id}`);
  }
}
```

Register the command in the CLI commander setup (wherever commands are registered):

```typescript
// In the candidates command group setup:
program
  .command('apply-resolution <candidateId>')
  .description('Apply the stored manual resolution to publish or merge a candidate')
  .option('--json', 'Output raw JSON response')
  .action(async (candidateId, options) => {
    const token = process.env.TRAPMAP_TOKEN;
    if (!token) {
      console.error('Error: TRAPMAP_TOKEN environment variable required');
      process.exit(1);
    }
    await applyResolution({
      candidateId,
      token,
      json: options.json,
    });
  });
```
