# Plan 56-02: Skill Feedback Prompts Parsing

> Extend SKILL.md frontmatter parsing to support optional feedbackPrompts field for skill-mounted feedback capabilities.

## Context

This plan extends the skill markdown parsing to support custom feedback prompts defined in SKILL.md frontmatter. Skill authors can define custom questions that appear after the standard feedback prompts when users report problems.

Example frontmatter:
```markdown
---
name: docker-cleanup-strategy
description: Clean up Docker resources safely
feedbackPrompts:
  - prompt: "Which Docker version are you using?"
    required: false
  - prompt: "What error message did you see?"
    required: true
---
```

export interface ParsedSkillMarkdown {

3. Add helper function `readFeedbackPrompts` after `readLabels`:
```typescript
function readFeedbackPrompts(value: unknown): FeedbackPrompt[] | undefined {
  if (!Array.isArray(value)) {
    return undefined;
  }

  const prompts: FeedbackPrompt[] = [];

  for (const item of value) {
    if (typeof item === 'object' && item !== null && 'prompt' in item) {
      const obj = item as Record<string, unknown>;
      const promptText = obj.prompt;
      if (typeof promptText === 'string' && promptText.trim().length > 0) {
        prompts.push({
          prompt: promptText.trim(),
          required: Boolean(obj.required ?? false),
        });
      }
    }
  }

  return prompts.length > 0 ? prompts : undefined;
}
```

4. Update `parseSkillMarkdown` function to include feedbackPrompts:
```typescript
export function parseSkillMarkdown(content: string): ParsedSkillMarkdown {
  const parsed = parseMarkdownFrontmatter(content);
  const data = parsed.data;
  const name = readString(data.name);
  const explicitTitle = readString(data.title);

  return {
    name,
    title: explicitTitle ?? name,
    description: readString(data.description),
    labels: readLabels(data.labels),
    body: parsed.body,
    hasFrontmatter: parsed.hasFrontmatter,
    feedbackPrompts: readFeedbackPrompts(data.feedbackPrompts),
  };
}
```

5. Export the FeedbackPrompt type (add to end of file or near the interface):
```typescript
export type { FeedbackPrompt };
```
