<action>
1. Create `packages/server/src/routes/admin-boundary-search.ts`:
   ```typescript
   import {
     adminBoundarySearchQuerySchema,
     adminBoundarySearchResponseSchema,
   } from '@trapmap/contracts';
   import type { SkillShareerServices, ResolvedAuthContext } from '../context.js';
   import { AppError } from '../errors.js';
   import { findEntriesByBoundaryConstraint } from '../lib/retrieval/boundary-query.js';

   /**
    * POST /admin/boundary-search
    * Find knowledge entries matching boundary constraints.
    * Requires system admin authentication.
    */
   export async function handleAdminBoundarySearch(
     services: SkillShareerServices,
     auth: ResolvedAuthContext,
     body: unknown,
   ): Promise<z.infer<typeof adminBoundarySearchResponseSchema>> {
     // Require system admin
     if (auth.subjectType !== 'system-admin') {
       throw new AppError(403, 'forbidden', 'System admin required');
     }

     // Validate query
     const query = adminBoundarySearchQuerySchema.parse(body);

     // Get store snapshot
     const data = await services.store.snapshot();

     // Build constraint from query
     const constraint = {
       context: query.context,
       platform: query.platform,
       package: query.package,
     };

     // Skip if no constraints provided
     if (!constraint.context && !constraint.platform && !constraint.package) {
       return {
         matches: [],
         query,
       };
     }

     // Find matching entries
     const matches = findEntriesByBoundaryConstraint(
       data.knowledgeEntries,
       constraint,
     ).slice(0, query.maxResults);

     // Build response
     return adminBoundarySearchResponseSchema.parse({
       matches: matches.map((entry) => ({
         entryId: entry.id,
         scope: entry.scope,
         shortcut: entry.shortcut,
         detail: entry.detail,
         labels: entry.labels,
         boundary: entry.boundary ?? null,
       })),
       query,
     });
   }
   ```
2. Ensure route is properly typed with Zod import
</action>
