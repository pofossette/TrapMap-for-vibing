   ```typescript
   import { describe, it, expect } from 'vitest';
   import {
     retrievalQuerySchema,

   } from './retrieval.js';

   describe('Retrieval schema boundary fields', () => {
     describe('retrievalQuerySchema.boundaryContext', () => {

         expect(result.boundaryContext?.versions).toHaveLength(1);

         expect(result.boundaryContext).toBeUndefined();

         });

         expect(result.boundaryContext?.contexts).toBeUndefined();
       });
     });

     describe('retrievalMatchSchema.boundaryExplanation', () => {

           labels: ['test'],

           reason: 'semantic similarity',
         });

       });
     });
   });
   ```

   ```typescript
   import { describe, it, expect, beforeEach } from 'vitest';
   import { searchKnowledge } from '../lib/retrieval/orchestrator.js';
   import type { SkillShareerServices, ResolvedAuthContext } from '../context.js';
   import { createTestServices, seedKnowledgeEntry } from '../test/utils.js';

   describe('Boundary-aware retrieval E2E', () => {
     let services: SkillShareerServices;
     let auth: ResolvedAuthContext;

     beforeEach(async () => {
       services = await createTestServices();
       auth = {
         actorId: 'user-1',
         subjectType: 'user',
         securityLevel: 'standard',
         activeTeamId: null,
       };

       // Seed test entries with different boundaries
       await seedKnowledgeEntry(services, {
         id: 'entry-1',
         shortcut: 'react-18-trap',
         detail: 'React 18 specific knowledge',
         labels: ['react'],
         scope: 'global',
         lifecycleState: 'approved',
         boundary: {
           context: ['frontend'],
           versions: [{ package: 'react', range: '>=18.0.0' }],
           prerequisites: [],
           signals: [],
           exclusions: [],
           evidence: [],
         },
       });

       await seedKnowledgeEntry(services, {
         id: 'entry-2',
         shortcut: 'node-trap',
         detail: 'Node.js knowledge',
         labels: ['nodejs'],
         scope: 'global',
         lifecycleState: 'approved',
         boundary: {
           context: ['backend'],
           versions: [{ package: 'node', range: '>=16.0.0' }],
           prerequisites: [],
           signals: [],
           exclusions: [{ description: 'Not for Windows', kind: 'platform' }],
           evidence: [],
         },
       });
     });

     it('includes boundaryExplanation in response when boundaryContext provided', async () => {
       const response = await searchKnowledge(services, auth, {
         seed: 'react hooks',
         boundaryContext: {
           contexts: ['frontend'],
           versions: [{ package: 'react', version: '18.2.0' }],
         },
       });

       // At least one match should have boundaryExplanation
       const allMatches = [...response.globalConstraints, ...response.projectKnowledge];
       const withExplanation = allMatches.filter(m => m.boundaryExplanation);

       expect(withExplanation.length).toBeGreaterThan(0);
       expect(withExplanation[0].boundaryExplanation?.checked).toBe(true);
     });

     it('excludes entry with unsatisfied version constraint', async () => {
       const response = await searchKnowledge(services, auth, {
         seed: 'react',  // matches both entries
         boundaryContext: {
           versions: [{ package: 'react', version: '17.0.0' }],  // Does NOT satisfy >=18.0.0
         },
       });

       const allMatches = [...response.globalConstraints, ...response.projectKnowledge];

       // entry-1 should be excluded (react >=18 required, we have 17)
       expect(allMatches.find(m => m.entryId === 'entry-1')).toBeUndefined();
     });

     it('penalizes entry with matching exclusion', async () => {
       const response = await searchKnowledge(services, auth, {
         seed: 'node backend',
         boundaryContext: {
           platform: 'windows',  // Matches exclusion on entry-2
         },
       });

       const allMatches = [...response.globalConstraints, ...response.projectKnowledge];
       const nodeEntry = allMatches.find(m => m.entryId === 'entry-2');

       if (nodeEntry) {
         // Entry should have warning in boundaryExplanation
         expect(nodeEntry.boundaryExplanation?.warnings.length).toBeGreaterThan(0);
       }
     });

     it('boosts entry with matching context', async () => {
       const response = await searchKnowledge(services, auth, {
         seed: 'frontend development',
         boundaryContext: {
           contexts: ['frontend'],  // Matches context on entry-1
         },
       });

       const allMatches = [...response.globalConstraints, ...response.projectKnowledge];
       const reactEntry = allMatches.find(m => m.entryId === 'entry-1');

       if (reactEntry) {
         // Entry should have boost in boundaryExplanation
         expect(reactEntry.boundaryExplanation?.boosts.length).toBeGreaterThan(0);
         expect(reactEntry.boundaryExplanation?.boosts[0]).toContain('Applicable context');
       }
     });
   });
   ```
