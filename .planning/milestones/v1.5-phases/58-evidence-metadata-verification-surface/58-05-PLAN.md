/**
 * Extract compact evidence hint from a knowledge entry or artifact record.
 * Returns null if the record has no evidence metadata.
 */
function extractEvidenceHint(record: {
  evidenceMeta: { sourceType: string; evidenceLevel: string; verifiedAt: string } | null;
}): EvidenceHint | null {
  if (!record.evidenceMeta) {
    return null;
  }
  return {
    evidenceLevel: record.evidenceMeta.evidenceLevel,
    verifiedAt: record.evidenceMeta.verifiedAt,
    sourceType: record.evidenceMeta.sourceType,
  };
}
```

import type { FastifyInstance } from 'fastify';
import { evidenceMetaSchema } from '@trapmap/contracts';

export async function evidenceRoutes(app: FastifyInstance) {
  app.patch<{
    Params: { id: string };
    Body: { sourceType?: string; sourceRef?: string; evidenceLevel?: string };
  }>('/v1/knowledge/:id/evidence', async (request, reply) => {
    const { id } = request.params;
    const entry = app.skillShareer.store.getKnowledgeEntry(id);
    if (!entry) {
      return reply.status(404).send({ error: 'Entry not found' });
    }

    const partialEvidence = request.body;
    const now = new Date().toISOString();
    const reviewerActorRef = /* derive from request auth context */;

    // Merge with existing or create new
    entry.evidenceMeta = {
      sourceType: partialEvidence.sourceType ?? entry.evidenceMeta?.sourceType ?? 'internal-experience',
      sourceRef: partialEvidence.sourceRef ?? entry.evidenceMeta?.sourceRef ?? '',
      evidenceLevel: partialEvidence.evidenceLevel ?? entry.evidenceMeta?.evidenceLevel ?? 'anecdotal',
      verifiedAt: now,
      verifiedBy: reviewerActorRef,
    };

    // Validate with zod schema
    evidenceMetaSchema.parse(entry.evidenceMeta);

    return { evidence: entry.evidenceMeta };
  });
}
```
